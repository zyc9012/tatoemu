# Function to apply common settings to the target
# This should be called after add_executable() in platform-specific CMakeLists.txt
function(apply_common_target_settings target_name)
    # Common include directories
    target_include_directories(${target_name} PRIVATE 
        ${CMAKE_SOURCE_DIR}/emulators
        ${CMAKE_SOURCE_DIR}/emulators/components/cpu/m68k
        ${CMAKE_SOURCE_DIR}/emulators/components/cpu/z80
        ${CMAKE_SOURCE_DIR}/emulators/components/sound
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/ym2151
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/msm6295
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/qsound
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/fm
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/ay8910
        ${CMAKE_SOURCE_DIR}/emulators/components/eeprom
        ${CMAKE_SOURCE_DIR}/utilities
    )

    # Add miniz, m68k, Z80, and sound component source files and compile them as C
    # Note: z80.cpp and z80daisy.cpp are C++ code, so they should be compiled as CXX
    # Note: z80ctc.cpp and z80pio.cpp have been removed - not used by CPS/NeoGeo emulators
    set_source_files_properties(
        ${CMAKE_SOURCE_DIR}/utilities/miniz/miniz.c
        ${CMAKE_SOURCE_DIR}/emulators/components/cpu/m68k/m68kcpu.c
        ${CMAKE_SOURCE_DIR}/emulators/components/cpu/m68k/m68kops.c
        ${CMAKE_SOURCE_DIR}/emulators/components/cpu/m68k/m68kcb.c
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/fm/fm.c
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/fm/ymdeltat.c
        ${CMAKE_SOURCE_DIR}/emulators/components/sound/ay8910/ay8910.c
        PROPERTIES LANGUAGE C
    )
    
    # Build minimal miniz
    set_source_files_properties(
        ${CMAKE_SOURCE_DIR}/utilities/miniz/miniz.c
        PROPERTIES COMPILE_DEFINITIONS "MINIZ_NO_DEFLATE_APIS;MINIZ_NO_TIME;MINIZ_NO_ARCHIVE_WRITING_APIS;MINIZ_NO_ZLIB_APIS;MINIZ_NO_ZLIB_COMPATIBLE_NAMES"
    )

    # Common compiler warnings
    target_compile_options(${target_name} PRIVATE 
        -Wall
        -Wextra
        -Wpedantic
    )
endfunction()
